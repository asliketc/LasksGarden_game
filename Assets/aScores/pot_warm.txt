rtsetparams(44100, 2)
load("MBLOWHOLE")
bus_config("MBLOWHOLE", "out 0-1")

notearray = {8.00, 8.04, 8.07, 8.11, 9.02}  
ampenv = maketable("line", 1000, 0,1, $2-0.1,1, $2,0)
start = 0

for (i = 0; i < 6; i = i + 1) {
    pch = notearray[i]
    freq = cpspch(pch)
    MBLOWHOLE(start, 0.4, 8000 * ampenv, freq*0.5, 0.25, 0.6, 0.5, irand(0,1), $4)
    start = start + 0.25
}
