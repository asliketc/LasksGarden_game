rtsetparams(44100, 2)
load("MSAXOFONY")
bus_config("MSAXOFONY", "out 0-1")

notearray = {8.05, 8.07, 8.09}  
ampenv = maketable("line", 1000, 0,1, $2-0.1,1, $2,0)
start = 0

for (i = 0; i < 5; i = i + 1) {
    pch = notearray[i]
    freq = cpspch(pch)
    MSAXOFONY(start, 0.3, 10000 * ampenv, freq, 0.3, 0.6, 0.5, 0.3, 0.5, $4)
    start = start + 0.2
}
